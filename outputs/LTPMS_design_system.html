SELECT c.course_code, c.course_name, u.real_name AS teacher,
       l.lab_code, c.major, COUNT(e.id) AS project_count
  FROM course c
  JOIN schedule s ON s.course_code = c.course_code AND s.semester_id = c.semester_id
  JOIN user u ON u.id = c.teacher_id
  JOIN laboratory l ON l.lab_code = s.lab_code
  LEFT JOIN experiment e ON e.course_id = c.id
 WHERE c.semester_id = 5
   AND l.lab_code = '36-401'
   AND c.teacher_id = :current_user_id -- 权限注入
 GROUP BY c.id
 ORDER BY c.course_code
 LIMIT 500;
